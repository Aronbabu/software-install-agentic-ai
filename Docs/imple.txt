import time
import winrm

from app.executors.result_models import ExecutionResult
from app.services.credential_provider import get_windows_credentials


def run_windows(job):
    creds = get_windows_credentials()

    host = job.target_host
    username = creds["username"]
    password = creds["password"]
    transport = creds["transport"]

    start = time.time()

    # Example package action — replace with controlled command builder
    ps_script = f"""
    Write-Output "Installing {job.software_name} {job.software_version or ''}"
    """

    session = winrm.Session(
        target=host,
        auth=(username, password),
        transport=transport,
        server_cert_validation="ignore"  # tighten for production certificate trust
    )

    response = session.run_ps(ps_script)
    duration = round(time.time() - start, 2)

    return ExecutionResult(
        success=(response.status_code == 0),
        exit_code=response.status_code,
        stdout=response.std_out.decode(errors="ignore") if response.std_out else "",
        stderr=response.std_err.decode(errors="ignore") if response.std_err else "",
        duration_seconds=duration,
        transport="winrm",
    )

Step D — Real Linux executor (Paramiko path)
app/executors/linux_executor.py

import time
import paramiko

from app.executors.result_models import ExecutionResult
from app.services.credential_provider import get_linux_credentials


def run_linux(job):
    creds = get_linux_credentials()

    host = job.target_host
    username = creds["username"]
    password = creds["password"]
    private_key_path = creds["private_key_path"]

    start = time.time()

    client = paramiko.SSHClient()
    client.load_system_host_keys()
    # For production, prefer known_hosts / explicit host keys.
    # Do NOT switch to AutoAddPolicy unless explicitly accepted for controlled environments.

    connect_kwargs = {
        "hostname": host,
        "username": username,
        "timeout": job.timeout_seconds or 300,
    }

    if private_key_path:
        key = paramiko.RSAKey.from_private_key_file(private_key_path)
        connect_kwargs["pkey"] = key
    else:
        connect_kwargs["password"] = password

    client.connect(**connect_kwargs)

    command = f"echo Installing {job.software_name} {job.software_version or ''}"
    stdin, stdout, stderr = client.exec_command(command)

    exit_code = stdout.channel.recv_exit_status()
    out = stdout.read().decode(errors="ignore")
    err = stderr.read().decode(errors="ignore")

    client.close()

    duration = round(time.time() - start, 2)

    return ExecutionResult(
        success=(exit_code == 0),
        exit_code=exit_code,
        stdout=out,
        stderr=err,
        duration_seconds=duration,
        transport="ssh",
    )

Why this direction
Paramiko documents the high-level pattern of SSHClient, load_system_host_keys(), connect(...), and exec_command(...), and it documents host-key handling instead of treating unknown hosts as trusted by default
