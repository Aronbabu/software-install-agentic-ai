services:

  postgres:
    image: postgres:16
    container_name: software-install-postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: software_install
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
    ports:
      - "5432:5432"
    volumes:
      - pgdata:/var/lib/postgresql/data

  redis:
    image: redis:7
    container_name: software-install-redis
    restart: unless-stopped
    ports:
      - "6379:6379"
    volumes:
      - redisdata:/data
    command: redis-server --appendonly yes

volumes:
  pgdata:
  redisdata:

docker compose up -d

docker ps

docker exec -it software-install-redis redis-cli ping
