2. Retry Refactor
services/execution_service.py

@celery_app.task(bind=True)
def process_job_task(self, job_id):

    try:
        process_job(job_id)

    except Exception as exc:

        if self.request.retries >= 3:
            raise

        raise self.retry(
            exc=exc,
            countdown=60
        )
