@router.post("/{job_id}/execute")
def execute(job_id: str, db: Session = Depends(get_db)):
    job = db.query(Job).filter(Job.id == job_id).first()

    if not job:
        raise HTTPException(status_code=404, detail="Job not found")

    # Step 1 → VALIDATING
    if job.status == JobStatus.PENDING:
        update_job_status(db, job, JobStatus.VALIDATING, "Validation started")

    # Step 2 → RUNNING
    if job.status == JobStatus.VALIDATING:
        update_job_status(db, job, JobStatus.RUNNING, "Execution started")

    result = execute_job(db, job)

    # ✅ NEW Step → VERIFYING
    if job.status == JobStatus.RUNNING:
        update_job_status(db, job, JobStatus.VERIFYING, "Verification started")

    # ✅ FINAL Step → SUCCESS / FAILED
    if result == "SUCCESS":
        update_job_status(db, job, JobStatus.SUCCESS, "Execution completed")
    else:
        update_job_status(db, job, JobStatus.FAILED, "Execution failed")
   
    if job.status in [JobStatus.SUCCESS, JobStatus.FAILED]:
     raise HTTPException(status_code=400, detail="Job already completed")


    return job
